Role-based navigation (main.dart)
// Auth check — user ka role dekho, route karo Future<void> _handleAuthState(User? user) async { if (user == null) { Get.offAllNamed(AppRoutes.login); return; } final doc = await FirebaseFirestore.instance .collection('users').doc(user.uid).get(); final role = doc['role']; final verified = doc['is_verified'] ?? false; if (role == 'admin') { Get.offAllNamed(AppRoutes.adminDashboard); } else if (role == 'mr' && verified) { Get.offAllNamed(AppRoutes.mrDashboard); } else { Get.offAllNamed(AppRoutes.pendingVerification); } }

Firestore security rules (important!)
rules_version = '2'; service cloud.firestore { match /databases/{db}/documents { // MR apna data hi dekh sakta hai match /visits/{id} { allow read: if request.auth.uid == resource.data.mr_id || isAdmin(); allow create: if request.auth.uid == request.resource.data.mr_id; } // Admin hi products manage kar sakta hai match /products/{id} { allow read: if isVerifiedMR() || isAdmin(); allow write: if isAdmin(); } function isAdmin() { return get(/databases/$(db)/documents/users/$(request.auth.uid)) .data.role == 'admin'; } function isVerifiedMR() { let u = get(/databases/$(db)/documents/users/$(request.auth.uid)).data; return u.role == 'mr' && u.is_verified == true; } } }
Important: GPS location ko visit confirm hone se pehle capture karo. Agar location permission deny ho, visit log allow mat karo — yahi anti-fraud protection hai.
